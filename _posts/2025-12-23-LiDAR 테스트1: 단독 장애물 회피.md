# **Mode 1) LiDAR Only Obstacle Avoidance**

---

## 동작 원리

> 라이다 센서로부터 받은 거리 데이터를 분석하여 좌우 방향의 최소 거리를 계산하고, 장애물의 위치에 따라 적절한 회피 동작을 수행
>

## 동작 확인


```python
# 새 터미널에서 라이다 기능 진입
ros2 service call /lidar_app/enter std_srvs/srv/Trigger {}

# Mode 1 시작
ros2 service call /lidar_app/set_running interfaces/srv/SetInt64 "{data: 1}"

# 기능 종료
ros2 service call /lidar_app/set_running interfaces/srv/SetInt64 "{data: 0}"

# 완전 종료
ros2 service call /lidar_app/exit std_srvs/srv/Trigger {}
```

## 코드 분석

### 일반 로봇 타입 (MentorPi A1 제외)

```python
def lidar_callback(self, lidar_data):
    ...
    with self.lock:  # 멀티스레드 환경에서 안전한 데이터 접근을 위한 락
        ...
        if self.machine_type != 'MentorPi_Acker':
            # 장애물 회피 모드이고, 이전 회전 동작이 완료된 경우에만 실행
            if self.running_mode == 1 and self.timestamp <= time.time():
                # 0이 아닌 값들의 인덱스 추출 (유효한 거리 데이터만 선택)
                left_nonzero = left_range.nonzero()
                right_nonzero = right_range.nonzero()
                # 유한한 값(inf, nan 제외)만 필터링
                left_nonan = np.isfinite(left_range[left_nonzero])
                right_nonan = np.isfinite(right_range[right_nonzero])
                # 유효한 거리 데이터만 추출
                min_dist_left_ = left_range[left_nonzero][left_nonan]
                min_dist_right_ = right_range[right_nonzero][right_nonan]
                
                # 좌우 최소 거리 계산 (최소 2개 이상의 데이터가 있어야 신뢰성 확보)
                if len(min_dist_left_) > 1 and len(min_dist_right_) > 1:
                    min_dist_left = min_dist_left_.min()   # 좌측 최소 거리
                    min_dist_right = min_dist_right_.min()  # 우측 최소 거리
                    
                    # 케이스 1: 좌측에만 장애물이 있는 경우 → 우회전
                    if min_dist_left <= self.threshold and min_dist_right > self.threshold:
                        twist.linear.x = self.speed / 6  # 회전 시 전진 속도를 최소화 (안정성 향상)
                        max_angle = math.radians(90)  # 90도 회전
                        w = self.speed * 6.0  # 회전 속도를 최대화 (빠른 회피)
                        twist.angular.z = -w  # 음수 = 우회전
                        # 이전 동작이 좌회전이었다면 방향 전환 (진동 방지)
                        if self.last_act != 0 and self.last_act != 1:
                            twist.angular.z = w
                        self.last_act = 1  # 현재 동작: 좌측 장애물 회피
                        self.mecanum_pub.publish(twist)
                        # 회전 완료 시간 계산: 각도 / 각속도 / 2 (절반 시간만 사용하여 빠른 반응)
                        self.timestamp = time.time() + (max_angle / w / 2)
                    
                    # 케이스 2: 양측 모두 장애물이 있는 경우 → 180도 회전 (후진)
                    elif min_dist_left <= self.threshold and min_dist_right <= self.threshold:
                        twist.linear.x = self.speed / 6  # 회전 시 전진 속도를 최소화
                        w = self.speed * 6.0  # 회전 속도를 최대화
                        twist.angular.z = w  # 양수 = 좌회전 (180도 회전)
                        self.last_act = 3  # 현재 동작: 180도 회전
                        self.mecanum_pub.publish(twist)
                        # 180도 회전 완료 시간 계산
                        self.timestamp = time.time() + (math.radians(180) / w / 2)
                    
                    # 케이스 3: 우측에만 장애물이 있는 경우 → 좌회전
                    elif min_dist_left > self.threshold and min_dist_right <= self.threshold:
                        twist.linear.x = self.speed / 6  # 회전 시 전진 속도를 최소화
                        max_angle = math.radians(90)  # 90도 회전
                        w = self.speed * 6.0  # 회전 속도를 최대화
                        twist.angular.z = w  # 양수 = 좌회전
                        # 이전 동작이 우회전이었다면 방향 전환 (진동 방지)
                        if self.last_act != 0 and self.last_act != 2:
                            twist.angular.z = -w
                        self.last_act = 2  # 현재 동작: 우측 장애물 회피
                        self.mecanum_pub.publish(twist)
                        # 회전 완료 시간 계산
                        self.timestamp = time.time() + (max_angle / w / 2)
                    
                    # 케이스 4: 장애물이 없는 경우 → 직진
                    else:
                        self.last_act = 0  # 현재 동작: 직진
                        twist.linear.x = self.speed  # 기본 속도로 직진
                        self.mecanum_pub.publish(twist)
```

- **동작 로직**
    1. **좌우 최소 거리 계산**: 유효한 거리 데이터에서 좌우 각각의 최소값을 구함
    2. **장애물 위치 판단**:
        - **좌측에만 장애물**: 우회전 (90도 회전)
        - **양측 모두 장애물**: 180도 회전
        - **우측에만 장애물**: 좌회전 (90도 회전)
        - **장애물 없음**: 직진
    3. **타임스탬프 관리**: 회전 시간 동안 다음 동작을 막아 회전이 완료되도록 함

### MentorPi A1 타입 로봇

```python
def lidar_callback(self, lidar_data):
    ...
    with self.lock:  # 멀티스레드 환경에서 안전한 데이터 접근을 위한 락
        ...
        else:  # MentorPi_Acker 타입 (아커만 스티어링 방식)
            # 장애물 회피 모드이고, 이전 회전 동작이 완료된 경우에만 실행
            if self.running_mode == 1 and self.timestamp <= time.time():
                # 유효한 거리 데이터 추출 (일반 타입과 동일)
                left_nonzero = left_range.nonzero()
                right_nonzero = right_range.nonzero()
                left_nonan = np.isfinite(left_range[left_nonzero])
                right_nonan = np.isfinite(right_range[right_nonzero])
                min_dist_left_ = left_range[left_nonzero][left_nonan]
                min_dist_right_ = right_range[right_nonzero][right_nonan]
                
                if len(min_dist_left_) > 1 and len(min_dist_right_) > 1:
                    min_dist_left = min_dist_left_.min()
                    min_dist_right = min_dist_right_.min()
                    
                    # 상태 머신: 이전 동작에 따라 다음 동작 결정 (복잡한 회피 시퀀스)
                    # 상태 3: 후진 후 회전 준비 단계
                    if self.last_act == 3: 
                        twist.linear.x = self.speed
                        # 아커만 스티어링: 각속도 = 선속도 / 회전반경 (0.1m)
                        twist.angular.z = -twist.linear.x/0.1  # 우회전, 회전 반경 0.1m로 설정
                        self.last_act = 4  # 다음 상태로 전환
                        self.mecanum_pub.publish(twist)
                        self.timestamp = time.time() + 0.5  # 0.5초 후 다음 단계
                    
                    # 상태 4: 중간 단계 (회전 완료 대기)
                    elif self.last_act == 4:
                        self.last_act = 5  # 다음 상태로 전환
                        self.mecanum_pub.publish(twist)
                        self.timestamp = time.time() + 0.1  # 짧은 대기 시간
                    
                    # 케이스 1: 좌측에만 장애물 → 우회전 (아커만 스티어링)
                    elif (min_dist_left <= self.threshold and min_dist_right > self.threshold):
                        twist.linear.x = self.speed
                        twist.angular.z = -twist.linear.x/0.1  # 우회전, 회전 반경 0.1m
                        self.last_act = 1
                        self.mecanum_pub.publish(twist)
                        self.timestamp = time.time() + 1
                    
                    # 케이스 2: 양측 모두 장애물 → 후진 + 서보 제어
                    elif min_dist_left <= self.threshold and min_dist_right <= self.threshold:
                        twist.linear.x = -0.3  # 후진 속도
                        # 서보 모터 제어: 서보 ID 3번을 중앙 위치(1500)로 설정
                        servo_state = PWMServoState()
                        servo_state.id = [3]  # 서보 ID
                        servo_state.position = [1500]  # 중앙 위치 (PWM 값)
                        data = SetPWMServoState()
                        data.state = [servo_state]
                        data.duration = 0.02  # 서보 동작 시간 (20ms)
                        self.servo_state_pub.publish(data)
                        self.last_act = 3  # 후진 상태로 전환
                        self.mecanum_pub.publish(twist)
                        self.timestamp = time.time() + 1
                        # 중복 코드 (안전성을 위한 이중 발행)
                        self.last_act = 3
                        self.mecanum_pub.publish(twist)
                        self.timestamp = time.time() + 1
                    
                    # 케이스 3: 우측에만 장애물
                    elif min_dist_left > self.threshold and min_dist_right <= self.threshold:
                        # 상태 5에서 온 경우: 후진 후 회전
                        if self.last_act == 5:
                            twist.linear.x = -0.3  # 후진
                            twist.angular.z = -twist.linear.x/0.1  # 회전
                            self.last_act = 3
                            self.mecanum_pub.publish(twist)
                            self.timestamp = time.time() + 1
                        # 일반적인 경우: 좌회전
                        else:
                            twist.linear.x = self.speed
                            twist.angular.z = -twist.linear.x/0.1  # 좌회전
                            self.last_act = 2
                            self.mecanum_pub.publish(twist)
                            self.timestamp = time.time() + 1
                    
                    # 케이스 4: 장애물 없음 → 직진 + 서보 중앙 복귀
                    else:
                        self.last_act = 0  # 직진 상태
                        twist.linear.x = 0.3  # 직진 속도
                        
                        # 서보 모터를 중앙 위치로 복귀 (아커만 스티어링 중립)
                        servo_state = PWMServoState()
                        servo_state.id = [3]  # 서보 ID
                        servo_state.position = [1500]  # 중앙 위치 (PWM 값)
                        data = SetPWMServoState()
                        data.state = [servo_state]
                        data.duration = 0.02  # 서보 동작 시간
                        self.servo_state_pub.publish(data)
                        
                        self.pid_yaw.clear()  # PID 상태 초기화
                        self.mecanum_pub.publish(twist)
```

- 서보 모터 제어와 더 정교한 회피 로직을 사용
- 특히 아커만 스티어링 기하학을 고려하여 회전 반경을 0.1m로 설정하고, 양측 모두 장애물이 있을 때는 후진과 서보 제어를 함께 수행
